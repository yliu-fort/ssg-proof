\begin{lemma}[The merged bound]\label{rl:merge-bound}
Let $G$ be a stopping SSG with $\Vmin=\emptyset$ and let $x\in R_1(G)$. Then $D:=x-w^{\ast}\ge0$ on $C$ and $D\le MD$.
\end{lemma}

\begin{theorem}[Merged transience forces level one]\label{rl:merge}
Let $G$ be stopping with $\Vmin=\emptyset$. If $M(G)$ is transient then $R_1(G)=\{w^{\ast}\}$.
\end{theorem}

\begin{theorem}[The level of a freezing set]\label{rl:levels}
Let $G$ be stopping with $\Vmin=\emptyset$, let $j\ge1$ and let $A\subseteq C$ with $|A|=j-1$. If $M^{A,p}$ is transient for every $p\in\{0,1\}^{A}$ then $R_j(G)=\{w^{\ast}\}$.
\end{theorem}

\begin{corollary}[Splitters bound the level]\label{rl:splitters}
Let $G$ be stopping with $\Vmin=\emptyset$. Then $R_j(G)=\{w^{\ast}\}$ for every $j>\operatorname{sp}(G)$. Contrapositively, if $\max_{R_j}x(v)>w^{\ast}(v)$ for some $v$ then $G$ has at least $j$ splitters.
\end{corollary}

\begin{proposition}[The cone relaxation certifies exactness]\label{rl:cone}
$\{x-w^{\ast}:x\in R_j(G)\}\subseteq\mathrm{CR}_j(G)$; so if $\mathrm{CR}_j(G)=\{0\}$ --- one linear programme --- then $R_j(G)=\{w^{\ast}\}$. The criterion is strictly stronger than \Cref{rl:levels}.
\end{proposition}

\begin{theorem}[Level one is asymptotically vacuous]\label{rl:tree-vacuous}
Let $n:=2^{d}$, $g=1-2^{-e}$, $S:=n\kappa+1-\kappa$, and assume $\kappa<g$ and $n(g-\kappa)\ge1-\kappa$. Then $x_\ell:=n\kappa(g+(1-g)2^{-\ell})/S$ (indexed by the depth $\ell$) lies in $R_1(T_d(e,\kappa))$, so $\max_{R_1}x(\mathrm{root})\ge n\kappa/(n\kappa+1-\kappa)=1-\Theta(1/N)$ while $w^{\ast}(\mathrm{root})=\kappa$ and $\max_{R_0}x(\mathrm{root})=1$.
\end{theorem}

\begin{corollary}[Level one does not decide the threshold, with one player]\label{rl:tree-threshold}
For $\kappa<\tfrac12$ and $2^{d}\kappa>1-\kappa$, $\min_{R_1}x(\mathrm{root})=\kappa<\tfrac12<\max_{R_1}x(\mathrm{root})$. The smallest instance is $T_2(3,\tfrac14)$: $14$ vertices, $7$ Max, $5$ average, $R_1$-interval $[\tfrac14,\tfrac47]$, $R_2$ exact at the root.
\end{corollary}
